## Builder

## Manager Methods

<TypeTable
  type={{
    "path(prefix)": {
      type: "IgniterStorageManager",
      description: "Append a path segment and return a new scoped manager",
    },
    "scope(key, identifier?)": {
      type: "IgniterStorageManager",
      description:
        "Bind to a registered scope; identifier required when path includes [identifier]",
    },
    "upload(file, destination, options?)": {
      type: "Promise<IgniterStorageFile>",
      description: "Upload Blob/File/Readable with optional replace strategy",
    },
    "uploadFromUrl(url, destination, options?)": {
      type: "Promise<IgniterStorageFile>",
      description: "Fetch a remote URL then upload to destination",
    },
    "uploadFromBuffer(buffer, destination, options?)": {
      type: "Promise<IgniterStorageFile>",
      description: "Upload bytes with optional contentType",
    },
    "uploadFromBase64(base64, destination, options?)": {
      type: "Promise<IgniterStorageFile>",
      description: "Upload base64 data; contentType optional",
    },
    "get(pathOrUrl)": {
      type: "Promise<IgniterStorageFile | null>",
      description: "Check existence and return metadata",
    },
    "list(prefix?)": {
      type: "Promise<IgniterStorageFile[]>",
      description: "List files under the current scope/base path",
    },
    "stream(pathOrUrl)": {
      type: "Promise<Readable>",
      description: "Open a readable stream for a file",
    },
    "delete(pathOrUrl)": {
      type: "Promise<void>",
      description: "Delete a file",
    },
    "copy(from, to)": {
      type: "Promise<IgniterStorageFile>",
      description: "Copy a file; adapter must support copy",
    },
    "move(from, to)": {
      type: "Promise<IgniterStorageFile>",
      description: "Move a file; adapter must support move",
    },
  }}
/>
